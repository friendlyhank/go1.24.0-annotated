📖 1. 包概述

strconv 包实现了基本数据类型与字符串表示之间的转换。它是Go标准库中最重要的包之一，提供了高效、安全的类型转换功能。

🔢 数值转换

支持整数、浮点数、布尔值与字符串之间的双向转换，包括不同进制的支持。

📝 字符串处理

提供字符串引用、转义、反引用等功能，支持ASCII和Unicode字符处理。

⚡ 高性能

采用优化算法，支持快速路径处理，最小化内存分配。

🛡️ 类型安全

严格的错误处理机制，防止溢出和格式错误。

💡 6. 使用示例

6.1 ParseBool - 布尔值解析

📝 ParseBool 基本用法示例
package main

import (
    "fmt"
    "strconv"
)

func main() {
    // 支持多种格式的布尔值
    examples := []string{"true", "false", "1", "0", "t", "f", "T", "F", "TRUE", "FALSE"}
    
    for _, s := range examples {
        if b, err := strconv.ParseBool(s); err == nil {
            fmt.Printf("ParseBool(%q) = %t\n", s, b)
        } else {
            fmt.Printf("ParseBool(%q) = error: %v\n", s, err)
        }
    }
}

// 输出:
// ParseBool("true") = true
// ParseBool("false") = false  
// ParseBool("1") = true
// ParseBool("0") = false
// ParseBool("t") = true
// ParseBool("f") = false

6.2 ParseInt - 整数解析（含符号和进制前缀）

🔢 ParseInt 完整功能演示
package main

import (
    "fmt"
    "strconv"
)

func main() {
    // 基本整数解析
    examples := []struct {
        str     string
        base    int
        bitSize int
    }{
        // 十进制数（含正负号）
        {"123", 10, 64},
        {"+456", 10, 64},
        {"-789", 10, 64},
        
        // 自动检测进制（base=0）
        {"0b1010", 0, 64},      // 二进制前缀
        {"0B1111", 0, 64},      // 二进制前缀（大写）
        {"0o755", 0, 64},       // 八进制前缀
        {"0x1A2B", 0, 64},      // 十六进制前缀
        {"0377", 0, 64},        // 传统八进制
        
        // 指定进制
        {"1010", 2, 64},        // 二进制
        {"755", 8, 64},         // 八进制
        {"1A2B", 16, 64},       // 十六进制
        {"ZZ", 36, 64},         // 36进制
        
        // 不同位大小
        {"127", 10, 8},         // int8 范围
        {"32767", 10, 16},      // int16 范围
        {"-2147483648", 10, 32}, // int32 最小值
    }
    
    for _, ex := range examples {
        if val, err := strconv.ParseInt(ex.str, ex.base, ex.bitSize); err == nil {
            fmt.Printf("ParseInt(%q, %d, %d) = %d\n", ex.str, ex.base, ex.bitSize, val)
        } else {
            fmt.Printf("ParseInt(%q, %d, %d) = error: %v\n", ex.str, ex.base, ex.bitSize, err)
        }
    }
}

// 输出示例:
// ParseInt("123", 10, 64) = 123
// ParseInt("+456", 10, 64) = 456
// ParseInt("-789", 10, 64) = -789
// ParseInt("0b1010", 0, 64) = 10
// ParseInt("0B1111", 0, 64) = 15
// ParseInt("0x1A2B", 0, 64) = 6699

6.4 Format 系列 - 类型转字符串

📤 Format 系列函数使用示例
package main

import (
    "fmt"
    "strconv"
)

func main() {
    // FormatBool
    fmt.Printf("FormatBool(true) = %q\n", strconv.FormatBool(true))
    fmt.Printf("FormatBool(false) = %q\n", strconv.FormatBool(false))
    
    // FormatInt - 不同进制
    num := int64(255)
    fmt.Printf("FormatInt(%d, 2) = %q\n", num, strconv.FormatInt(num, 2))   // 二进制
    fmt.Printf("FormatInt(%d, 8) = %q\n", num, strconv.FormatInt(num, 8))   // 八进制
    fmt.Printf("FormatInt(%d, 10) = %q\n", num, strconv.FormatInt(num, 10)) // 十进制
    fmt.Printf("FormatInt(%d, 16) = %q\n", num, strconv.FormatInt(num, 16)) // 十六进制
    
    // FormatFloat - 不同格式
    f := 3.14159265359
    fmt.Printf("FormatFloat(%.11f, 'f', 2, 64) = %q\n", f, strconv.FormatFloat(f, 'f', 2, 64))
    fmt.Printf("FormatFloat(%.11f, 'e', 2, 64) = %q\n", f, strconv.FormatFloat(f, 'e', 2, 64))
    fmt.Printf("FormatFloat(%.11f, 'g', -1, 64) = %q\n", f, strconv.FormatFloat(f, 'g', -1, 64))
}

// 输出:
// FormatBool(true) = "true"
// FormatBool(false) = "false"
// FormatInt(255, 2) = "11111111"
// FormatInt(255, 8) = "377"
// FormatInt(255, 10) = "255"
// FormatInt(255, 16) = "ff"
// FormatFloat(3.14159265359, 'f', 2, 64) = "3.14"
// FormatFloat(3.14159265359, 'e', 2, 64) = "3.14e+00"
// FormatFloat(3.14159265359, 'g', -1, 64) = "3.14159265359"

6.5 Append 系列 - 高效字符串构建

⚡ Append 系列高性能字符串构建
package main

import (
    "fmt"
    "strconv"
)

func main() {
    // 使用Append系列函数避免多次内存分配
    buf := make([]byte, 0, 100)
    
    buf = strconv.AppendBool(buf, true)
    buf = append(buf, ' ')
    buf = strconv.AppendInt(buf, -42, 10)
    buf = append(buf, ' ')
    buf = strconv.AppendFloat(buf, 3.14, 'f', 2, 64)
    buf = append(buf, ' ')
    buf = strconv.AppendQuote(buf, "hello world")
    
    fmt.Printf("结果: %s\n", buf)
    // 输出: 结果: true -42 3.14 "hello world"
}

6.6 Quote 系列 - 字符串引用和转义

🔤 Quote 系列字符串处理演示
package main

import (
    "fmt"
    "strconv"
)

func main() {
    examples := []string{
        "hello world",
        "line1\nline2",
        "tab\there",
        "quote\"test",
        "中文测试",
        "\x00\x01\x02", // 控制字符
    }
    
    for _, s := range examples {
        quoted := strconv.Quote(s)
        ascii := strconv.QuoteToASCII(s)
        fmt.Printf("原文: %q\n", s)
        fmt.Printf("Quote: %s\n", quoted)
        fmt.Printf("QuoteToASCII: %s\n", ascii)
        
        // 反引用
        if unquoted, err := strconv.Unquote(quoted); err == nil {
            fmt.Printf("Unquote: %q\n", unquoted)
        }
        fmt.Println("---")
    }
}

6.7 错误处理示例

❌ NumError 错误处理最佳实践
package main

import (
    "fmt"
    "strconv"
)

func main() {
    examples := []string{"abc", "999999999999999999999", "1.2.3", ""}
    
    for _, s := range examples {
        if val, err := strconv.ParseInt(s, 10, 64); err != nil {
            if numErr, ok := err.(*strconv.NumError); ok {
                fmt.Printf("解析错误 - 函数: %s, 输入: %q, 类型: %T, 原因: %v\n", 
                          numErr.Func, numErr.Num, numErr.Err, numErr.Err)
            }
        } else {
            fmt.Printf("ParseInt(%q) = %d\n", s, val)
        }
    }
}

// 输出示例:
// 解析错误 - 函数: ParseInt, 输入: "abc", 类型: *errors.errorString, 原因: invalid syntax
// 解析错误 - 函数: ParseInt, 输入: "999999999999999999999", 类型: *errors.errorString, 原因: value out of range
